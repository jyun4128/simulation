파동의 독립성과 중첩 원리

두 개의 초음파(음파)는 서로 마주 보고 진행하더라도 독립적으로 진행 방향을 유지합니다. 그러나 겹치는 순간, 파동의 매질 변위는 두 변위의 벡터 합과 같은 중첩 원리를 따릅니다.

센서 신호 혼선(Crosstalk)

차량 A에서 발사한 고유의 펄스 음파를 차량 B의 수신부(Echo)가 자신이 보낸 반사파로 착각하여 타이밍 연산을 조기 차단함으로써, 거리가 극도로 가깝거나(0) 비정상적인 값으로 측정되는 오류를 범하게 됩니다.

난수 기반 지연 (Random Delay) 회피

두 차량이 통신할 수 없는 상황에서 고정된 발사 주기를 가지면, 주기가 겹칠 때 연속적인 혼선이 발생합니다. 이를 막기 위해 다음 초음파를 쏠 때까지의 대기 시간(Delay)을 컴퓨터가 임의의 난수(예: 30~70ms)로 매번 다르게 설정합니다. 우연히 한 번 겹치더라도 다음번엔 주기가 어긋나므로 연속 혼선 확률이 0에 수렴합니다.

// 아두이노 보정 코드 - 난수 발생(Random Delay)을 통한 혼선 회피 알고리즘 
const int trigA = 2; // 내 차량 송신기
const int echoA = 3; // 내 차량 수신기
const int ledAlert = 13; // 충돌 알림 LED

void setup() {
  Serial.begin(9600);
  pinMode(trigA, OUTPUT); pinMode(echoA, INPUT);
  pinMode(ledAlert, OUTPUT);
  // 아날로그 핀의 노이즈를 읽어 완벽한 난수 시드(Seed) 생성
  randomSeed(analogRead(0)); 
}

void loop() {
  long duration, distance;

  // --- 초음파 발사 및 측정 ---
  digitalWrite(trigA, LOW); delayMicroseconds(2);
  digitalWrite(trigA, HIGH); delayMicroseconds(10);
  digitalWrite(trigA, LOW);
  duration = pulseIn(echoA, HIGH);
  distance = duration * 0.034 / 2; // 거리 계산 공식

  Serial.println(distance);

  if (distance > 0 && distance < 30) {
    digitalWrite(ledAlert, HIGH); // 경보 켬
  } else {
    digitalWrite(ledAlert, LOW);  // 경보 끔
  }

  // 🎲 핵심 포인트: 난수 기반 무작위 지연 (Random Delay)
  // 상대방 차량과 주기가 동기화되어 연속 혼선이 일어나는 것을 방지하기 위해
  // 30ms ~ 70ms 사이의 랜덤한 시간 동안 대기 후 다음 사이클을 시작함.
  long randomWait = random(30, 71); 
  delay(randomWait); 
}
